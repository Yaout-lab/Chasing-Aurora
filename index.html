function showTab(id){

  ['lichtrinh','chuanbi','homnay','trangphuc','dieukien'].forEach(x=>document.getElementById(x).classList.add('hidden'));

  document.getElementById(id).classList.remove('hidden');

  document.querySelectorAll('[data-tab]').forEach(b=>{

    b.classList.toggle('active', b.getAttribute('data-tab') === id);

  });

  window.scrollTo({top:0,behavior:'smooth'});

}


// Lưu trạng thái checklist vào trình duyệt

document.querySelectorAll('.check input[type="checkbox"]').forEach(cb => {

  const key = 'ru_trip_' + cb.dataset.key;

  cb.checked = localStorage.getItem(key) === 'true';

  cb.addEventListener('change', () => localStorage.setItem(key, cb.checked));

});


const target = new Date('2027-01-01T22:00:00+03:00');

function tick(){

  const diff = target - new Date();

  if(diff<0) return;

  document.getElementById('d').innerText = Math.floor(diff/86400000);

  document.getElementById('h').innerText = Math.floor(diff/3600000)%24;

  document.getElementById('m').innerText = Math.floor(diff/60000)%60;

  document.getElementById('s').innerText = Math.floor(diff/1000)%60;

}

setInterval(tick,1000); tick();

  3:'Sáng chơi Saint, 17h55 bay Saint - Murmansk, 20h30 check-in phòng bằng mã.',

  7:'8h trả phòng, 10h ra sân bay về.'

};

const isTripMonth = now.getFullYear() === 2027 && now.getMonth() === 0;

document.getElementById('todayLine').innerText = 'Hôm nay: ' + now.toLocaleDateString('vi-VN');

document.getElementById('homnayText').innerText = (isTripMonth && plans[now.getDate()]) || 'Chưa tới ngày đi, chuẩn bị đồ đi bà. Còn ' + Math.max(0, Math.floor((target-now)/86400000)) + ' ngày nữa.';
